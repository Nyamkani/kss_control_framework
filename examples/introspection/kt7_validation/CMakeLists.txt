cmake_minimum_required(VERSION 3.16)
project(kcf_kt7_multi_application_validation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(KCF_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(KCF_TOOLS_SOURCE_DIR "" CACHE PATH "Existing kcf_tools checkout (read only)")
if(NOT EXISTS "${KCF_TOOLS_SOURCE_DIR}/src/ui/main_window.cpp")
    message(FATAL_ERROR "KCF_TOOLS_SOURCE_DIR must point to the existing KT-7 source tree")
endif()
find_package(Threads REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_subdirectory("${KCF_SOURCE_DIR}/kcf" external/kcf)
add_executable(kt7_fixture "${KCF_TOOLS_SOURCE_DIR}/tests/real_fixture.cpp" "${KCF_SOURCE_DIR}/bringup/src/bringup.cpp")
target_include_directories(kt7_fixture PRIVATE "${KCF_SOURCE_DIR}/bringup/include")
target_link_libraries(kt7_fixture PRIVATE kcf_core)
add_executable(kt7_multi_application main.cpp
    "${KCF_TOOLS_SOURCE_DIR}/src/ui/main_window.cpp"
    "${KCF_TOOLS_SOURCE_DIR}/src/backend/kcf_backend.cpp"
    "${KCF_TOOLS_SOURCE_DIR}/src/backend/value_codec.cpp")
target_include_directories(kt7_multi_application PRIVATE "${KCF_TOOLS_SOURCE_DIR}/include")
target_link_libraries(kt7_multi_application PRIVATE Qt6::Widgets kcf_core)
foreach(target kt7_fixture kt7_multi_application)
    target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
enable_testing()
add_test(NAME kt7_real_multi_application COMMAND kt7_multi_application $<TARGET_FILE:kt7_fixture>)
set_tests_properties(kt7_real_multi_application PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
